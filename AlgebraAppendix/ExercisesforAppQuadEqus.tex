\begin{problem}
Prove that for every nonzero number $p$, $x^2 + xp + p^2 = 0$  has no real solutions.

\begin{solution}
The discriminant is: $D = p^2 - 4p^2 = -3p^2 < 0$.  Since $D < 0$, there are no real solutions.  
\end{solution}
\end{problem}

\begin{problem}
Solve for $t$: $-\frac{1}{2} g t^2 + vt + h = 0$. Assume $g > 0$, $v \geq 0$ and $h \geq 0$.

\begin{solution}
$t = \frac{v \pm \sqrt{v^2 + 2gh}}{g}$
\end{solution}
\end{problem}
