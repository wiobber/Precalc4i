\begin{question}
In Exercises \ref{factoringexfirst} - \ref{factoringexlast}, factor completely over the integers.  Check your answer by multiplication.

\begin{problem} \label{factoringexfirst}
$2x - 10x^2$ 

\begin{solution}
$2x(1 - 5x)$
\end{solution}
\end{problem}

\begin{problem}
$12t^5 - 8t^3= 4t^3\left(\answer{3t^2-2}\right)$
\end{problem}

\begin{problem}
$16xy^2 - 12x^2y$

\begin{solution}
$4xy(4y-3x)$
\end{solution}
\end{problem}

\begin{problem}
$5(m+3)^2- 4(m+3)^3$

\begin{solution}

$(m+3)^2(-4m-7) = -(m+3)^2(4m+7)$

\end{solution}
\end{problem}

\begin{problem}
$(2x-1)(x+3) - 4(2x-1) = (2x-1)\left(\answer{x-1}\right)$

\end{problem}

\begin{problem}
$t^2(t-5) + t - 5$

\begin{solution}
$(t-5)(t^2+1)$
\end{solution}
\end{problem}

\begin{problem}
$w^2 - 121 = (w-11) \left(\answer{w+11}\right)$
\end{problem}

\begin{problem}
$49 - 4t^2$

\begin{solution}
$(7-2t)(7+2t) = - (2t-7)(2t+7)$
\end{solution}
\end{problem}

\begin{problem}
$81t^4 - 16$

\begin{solution}
$(3t-2)(3t+2)(9t^2+4)$
\end{solution}
\end{problem}

\begin{problem}
$9z^2 - 64y^4 = (3z-8y^2)\left(\answer{3z+8y^2}\right)$
\end{problem}

\begin{problem}
$(y+3)^2 - 4y^2$

\begin{solution}
 $3(-y+3)(y+1) = -3(y - 3)(y+1)$
\end{solution}
\end{problem}

\begin{problem}
$(x+h)^3 - (x+h)$

\begin{solution}
$(x+h)(x+h-1)(x+h+1)$
\end{solution}
\end{problem}

\begin{problem}
$y^2 - 24y + 144 = \left(\answer{y-12}\right)^2$

\end{problem}

\begin{problem}
$25t^2 + 10t + 1$

\begin{solution}
$(5t+1)^2$
\end{solution}
\end{problem}

\begin{problem}
$12x^3 - 36x^2 + 27x$

\begin{solution}
$3x(2x-3)^2$
\end{solution}
\end{problem}

\begin{problem}
$m^4 + 10m^2 + 25 = \left(\answer{m^2+5}\right)^2$

\end{problem}

\begin{problem}
$27 - 8x^3$

\begin{solution}
 $(3-2x)(9 + 6x + 4x^2) = -(2x-3)(4x^2+6x+9)$
\end{solution}
\end{problem}

\begin{problem}
$t^6 +t^3$

\begin{solution}
$t^3(t+1)(t^2 - t + 1)$
\end{solution}
\end{problem}

\begin{problem}
$x^2 - 5x - 14 = (x-7)\left(\answer{x+2}\right)$

\end{problem}
 
\begin{problem}
$y^2 - 12y + 27$

\begin{solution}
$(y-9)(y-3)$
\end{solution}
\end{problem}

\begin{problem}
$3t^2 + 16t + 5 = \left(\answer{3t+1}\right)(t+5)$
\end{problem}

\begin{problem}
$6x^2 - 23x + 20$

\begin{solution}
$(2x-5)(3x-4)$
\end{solution}
\end{problem}

\begin{problem}
$35+2m - m^2$

\begin{solution}
$(7-m)(5+m) = -(m-7)(m+5)$
\end{solution}
\end{problem}

\begin{problem}
$7w - 2w^2 - 3$

\begin{solution}
$(-2w+1)(w-3) = -(2w-1)(w-3)$
\end{solution}
\end{problem}

\begin{problem}
$3m^3 + 9m^2 - 12m = 3m(m-1)\left(\answer{m+4}\right)$
\end{problem}

\begin{problem}
$x^4 + x^2 - 20$

\begin{solution}
$(x-2)(x+2)(x^2+5)$
\end{solution}
\end{problem}

\begin{problem}
$4(t^2-1)^2 +3(t^2-1) - 10$

\begin{solution}
$(2t-3)(2t+3)(t^2+1)$
\end{solution}
\end{problem}

\begin{problem}
$x^3 - 5x^2 - 9x + 45 = (x-3)(x+3)\left(\answer{x-5}\right)$

\end{problem}

\begin{problem}
$3t^2 + t - 3 - t^3$

\begin{solution}
$(t-3)(1-t)(1+t) = -(t-3)(t-1)(t+1)$
\end{solution}
\end{problem}

\begin{problem} \label{factoringexlast}
$y^4 + 5y^2 + 9$

\begin{hint}
$y^4 + 5y^2 + 9 = (y^4 + 6y^2 + 9) - y^2$
\end{hint}

\begin{solution}
 $(y^2-y+3)(y^2+y+3)$
\end{solution}
\end{problem}

\begin{question}
In Exercises \ref{solvebyfactorfirst} - \ref{solvebyfactorlast},  find all rational number solutions.  Check your answers.

\begin{problem} \label{solvebyfactorfirst}
$(7x+3)(x-5) = 0$

\begin{solution}
 $x = -\frac{3}{7}$ or $x = 5$ 
\end{solution}
\end{problem}

\begin{problem}
$(2t-1)^2 (t+4) = 0$

$t = \answer{-4}$  (smaller solution)

$t = \answer{\frac{1}{2}}$ (larger solution)

\end{problem}

\begin{problem}
$(y^2 + 4)(3y^2 +y - 10) = 0$

\begin{solution}
$y = \frac{5}{3}$ or $y = -2$
\end{solution}
\end{problem}

\begin{problem}
$4t = t^2$

$t = \answer{0}$ (smaller solution)

$t = \answer{4}$ (larger solution)

\end{problem}

\begin{problem}
$y+3 = 2y^2$

\begin{solution}
$y = -1$ or $y = \frac{3}{2}$ 
\end{solution}
\end{problem}

\begin{problem}
$26x = 8x^2 + 21$ 

 $x = \answer{\frac{3}{2}}$ (smaller solution)
 
$x = \answer{\frac{7}{4}}$ (larger solution)
\end{problem}

\begin{problem}
$16x^4 = 9x^2$

\begin{solution}
$x = 0$ or $x = \pm \frac{3}{4}$ 
\end{solution}

\end{problem}

\begin{problem}
$w(6w+11) = 10$

$w = \answer{-\frac{5}{2}}$ (smaller solution)

$w = \answer{\frac{2}{3}}$ (larger solution)
\end{problem}

\begin{problem}
$2w^2 + 5w + 2 = - 3(2w+1)$

\begin{solution}
$w=-5$ or $w = -\frac{1}{2}$
\end{solution}
\end{problem}

\begin{problem}
$x^2(x-3) = 16(x-3)$

\begin{solution}
 $x=3$ or $x = \pm 4$
\end{solution}
\end{problem}

\begin{problem}
$(2t+1)^3 = (2t+1)$

\begin{solution}
$t = -1$, $t= -\frac{1}{2}$, or $t = 0$
\end{solution}
\end{problem}

\begin{problem}
$a^4 + 4 = 6 - a^2$

\begin{solution}
$a = \pm 1$
\end{solution}
\end{problem}

\begin{problem}
$\frac{8t^2}{3} = 2t+3$

$t = \answer{-\frac{3}{4}}$ (smaller solution)

$t = \answer{\frac{3}{2}}$ (larger solution)

\end{problem}

\begin{problem}
$\frac{x^3+x}{2} = \frac{x^2+1}{3}$

$x = \answer{\frac{2}{3}}$

\end{problem}

\begin{problem} \label{solvebyfactorlast}
$\frac{y^4}{3} - y^2 = \frac{3}{2} (y^2 + 3)$ 

\begin{solution}
 $y = \pm 3$ 
\end{solution}
\end{problem}
