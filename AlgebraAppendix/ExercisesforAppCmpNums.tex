\begin{question}
In Exercises \ref{compnumbasicfirst} - \ref{compnumbasiclast}, use the given complex numbers $z$ and $w$ to find and simplify the following. 

\begin{itemize}

\item $z+w$
\item $zw$
\item $z^2$
\item $\frac{1}{z}$
\item $\frac{z}{w}$
\item $\frac{w}{z}$
\item $\overline{z}$
\item $z\overline{z}$
\item $(\overline{z})^2$
\end{itemize}

\begin{problem} \label{compnumbasicfirst}
$z = 2+3i$, $w = 4i$ 

\begin{solution}

\begin{itemize}
\item $z+w = 2+7i$

\item $zw = -12+8i$

\item $z^2 = -5 + 12i$

\item $\frac{1}{z} = \frac{2}{13} - \frac{3}{13} \, i$

\item $\frac{z}{w} = \frac{3}{4} - \frac{1}{2} \, i$

\item $\frac{w}{z} = \frac{12}{13} + \frac{8}{13} \,i$

\item $\overline{z} = 2-3i$

\item $z\overline{z} = 13$

\item $(\overline{z})^2 = -5-12i$

\end{itemize}

\end{solution}
\end{problem}

\begin{question}

We know that $i^{2} = -1$ which means $i^{3} = i^{2} \cdot i = (-1) \cdot i = -i$ and $i^{4} = i^{2} \cdot i^{2} = (-1)(-1) = 1$. In Exercises \ref{powerofifirst} - \ref{powerofilast}, use this information to simplify the given power of $i$.

\begin{problem} \label{powerofifirst}
$i^{5}$ 

\begin{solution}
$i^{5} = i^{4} \cdot i = 1 \cdot i = i$
\end{solution}
\end{problem}

\begin{problem}
$i ^{6}$

\begin{solution}
$i ^{6} =  i^{4} \cdot i^{2} = 1 \cdot (-1) = -1$
\end{solution}
\end{problem}

\begin{problem}
$i^{7} = \answer{-i}$
\end{problem}

\begin{problem}
$i^{8} = \answer{1}$
\end{problem}

\begin{problem} 
$i^{15}$

\begin{solution}
$i^{15} = \left(i^{4}\right)^{3} \cdot i^{3} = 1 \cdot (-i) = -i$
\end{solution}
\end{problem}

\begin{problem}
$i^{26} = \answer{-1}$

\end{problem}

\begin{problem}
$i^{117}$

\begin{solution}
$i^{117} = \left(i^{4}\right)^{29} \cdot i = 1 \cdot i = i$
\end{solution}
\end{problem}

\begin{problem} \label{powerofilast}
$i^{304} = \answer{1}$ 
\end{problem}

\end{question}
