\begin{problem} \label{tangentlinepolyfirst}  
$f(x) = x-x^2$ 

\begin{solution}
\begin{enumerate}  

\item $f'(2) = \lim_{h \rightarrow 0} (-h-3) = -3$

\item  $y = f'(2)(x-2) + f(2) = (-3)(x-2)+(-2)$ so $y  = -3x+4$.

\item   $f'(x) =  \lim_{h \rightarrow 0} (-2x-h+1) = -2x+1$ 

\item  $y = f'(0)(x-0) + f(0) = (1)(x-0) + 0$ so $y = x$.

\end{enumerate}
\end{solution}
\end{problem}

\begin{problem} \label{tangentlinepolylast} 
$f(x) = x^{3} + 1$

\begin{solution}
\begin{enumerate}

\item $f'(2) = \lim_{h \rightarrow 0} \left( h^2+6h+12 \right) = 12$

\item  $y = f'(2)(x-2) + f(2) = 12(x-2)+ 9$ so $y = 12x - 15$.

\item  $f'(x) = \lim_{h \rightarrow 0} \left(3x^{2} + 3xh + h^{2} \right) = 3x^{2}$ 

\item $y = f'(0)(x-0)+f(0) = 0(x-0)+1$ so $y = 1$.

\end{enumerate} 
\end{solution}
\end{problem}

If we zoom in near the the vertex of a parabola, the graph becomes locally flat so it makes sense the slope of the tangent line,  $f'(x) = 0$ there.

\end{solution}
\end{problem}

\begin{problem}
$f(x) = \frac{3}{1-x}$

\begin{solution}
\begin{enumerate}
\item $\lim_{\Delta x \rightarrow 0} \frac{-3}{2(\Delta x - 2)} = \frac{3}{4}$

\item $\lim_{\Delta x \rightarrow 0} \frac{3}{(x+\Delta x-1)(x-1)} =  \frac{3}{(x-1)^2}$
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
$f(x) = \frac{1}{x^2}$

\begin{solution}
\begin{enumerate}
\item $\lim_{\Delta x \rightarrow 0} \frac{2-\Delta x}{(\Delta x - 1)^2} =2$

\item $\lim_{\Delta x \rightarrow 0}\frac{-(2x+\Delta x)}{x^2(x+\Delta x)^2} =  - \frac{2}{x^3}$ 
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem} \label{diffquotexerlastb}
$f(x) = \frac{2}{x+5}$

\begin{solution}
\begin{enumerate}
\item $\lim_{\Delta x \rightarrow 0} \frac{-1}{2(\Delta x+4)} = - \frac{1}{8}$

\item $\lim_{\Delta x \rightarrow 0} \frac{-2}{(x+5)(x+\Delta x+5)}=  - \frac{2}{(x+5)^2}$
\end{enumerate}
\end{solution}
\end{problem}
  
\end{question}

\begin{problem}\label{rationaltangentfirst} 
$f(x) = \frac{1}{4x-3}$ 

\begin{solution}
\begin{enumerate}

\item  $f'(-1) =\lim_{\Delta x \rightarrow 0} \frac{4}{7(4 \Delta x - 7)} = - \frac{4}{49}$

\item $y = f'(-1)(x-(-1)) + f(-1) = -\frac{4}{49}(x + 1) + \left(-\frac{1}{7}\right)$ so $y = -\frac{4}{49} \, x - \frac{11}{49}$.   

\item  $f'(x) = \lim_{\Delta x \rightarrow 0} \frac{-4}{(4x-3)(4x+4\Delta x-3)} =  - \frac{4}{(4x-3)^2}$  

\item $y = f'(0)(x-0)+f(0) = -\frac{4}{9} (x-0) + \left(-\frac{1}{3}\right)$ so $y = -\frac{4}{9} \, x - \frac{1}{3}$

\end{enumerate}

\end{solution}
\end{problem}

\begin{problem}
 $f(x) = \frac{3x}{x+2}$

 \begin{solution}
\begin{enumerate}

\item  $f'(-1) = \lim_{\Delta x \rightarrow 0} \frac{6}{\Delta x + 1}= 6$

\item $y = f'(-1)(x-(-1)) + f(-1) = 6(x+1) + (-3)$ so $y = 6x+3$.

\item  $f'(x) =\lim_{\Delta x \rightarrow 0} \frac{6}{(x+2)(x+\Delta x+2)} =   \frac{6}{(x+2)^2}$   

\item  $y = f'(0)(x-0) + f(0) = \frac{3}{2} \, (x-0)+0$ so $y = \frac{3}{2} \, x$

\end{enumerate}
 \end{solution}
 \end{problem}

\begin{problem}
$f(x) = \frac{x}{x - 9}$

\begin{solution}
\begin{enumerate}

\item  $f'(-1) = \lim_{\Delta x \rightarrow 0} \frac{9}{10(\Delta x - 10)} = -\frac{9}{100}$

\item $y = f'(-1)(x-(-1)) + f(-1) = -\frac{9}{100} \, (x+1) + \frac{1}{10}$ so $y = - \frac{9}{100} \, x + \frac{1}{100}$.


\item  $f'(x) =\lim_{\Delta x \rightarrow 0} \frac{-9}{(x - 9)(x + \Delta x - 9)} =   -\frac{9}{(x-9)^2}$    

\item  $y = f'(0)(x-0) + f(0) = -\frac{1}{9} \, (x-0)+0$ so $y = -\frac{1}{9} \, x$

\end{enumerate}
\end{solution}
\end{problem}

\begin{problem} \label{rationaltangentlast} 
$f(x) = \frac{x^2}{2x+1}$ 

\begin{solution}
\begin{enumerate}  

\item  $f'(-1) = \lim_{\Delta x \rightarrow 0} \frac{\Delta x}{2 \Delta x - 1} = 0$

\item $y = f'(-1)(x-(-1)) + f(-1) = (0) (x+1) + (-1)$ so $y = -1$.

\item  $f'(x) =\lim_{\Delta x \rightarrow 0}\frac{2x^2+2x\Delta x+2x+\Delta x}{(2x+1)(2x+2\Delta x+1)}=  \frac{2x^2+2x}{(2x+1)^2}$

\item  $y = f'(0)(x-0) + f(0) = (0)(x-0)+0$ so $y = 0$

\end{enumerate}
\end{solution}
\end{problem}

\end{question}

\begin{problem} For  $g(t) = t \sqrt{t}$:

\begin{enumerate}

\item  Explain why $g'(0) = \lim_{\Delta t \rightarrow 0} \frac{g(\Delta t) - g(0)}{\Delta t}$ does not exist.

\begin{solution}
If $\Delta t < 0$, then $\sqrt{\Delta t}$ is not a real number.  Hence, $g'(0) = \lim_{\Delta t \rightarrow 0} \frac{g(\Delta t) - g(0)}{\Delta t}$ does not exist.
\end{solution}

\item Find the \index{derivative from the right}\index{derivative ! from the right} derivative \textbf{from the right} at $t=0$:  $g_{+}'(0) = \lim_{\Delta t \rightarrow 0^{+}} \frac{g(\Delta t) - g(0)}{\Delta t}$

\begin{solution}
 $g_{+}'(0) =\lim_{\Delta t \rightarrow 0^{+}} (\Delta t)^{\frac{1}{2}}   =  0$
\end{solution}

\item  Find $y = g_{+}'(0) (x-0) + g(0)$ and interpret.

\begin{solution}
 $y = g_{+}'(0) (x - 0) + g(0) =  (0) (x-0) + 0$ so $y = 0$

 This line is a tangent line to the graph of $y = t \sqrt{t}$ at $(0,0)$ for $t \geq 0$.
\end{solution}

\item Find  $g'(t) =  \lim_{\Delta t \rightarrow 0} \frac{g(t+\Delta t) - g(t)}{\Delta t}$.  Assume $t>0$.

\begin{solution}
$g'(t) =\lim_{\Delta t \rightarrow 0}\frac{3t^2+3t\Delta t+(\Delta t)^2}{(t+\Delta t)^{3/2} + t^{3/2}} = \frac{3t^2}{2 t^{3/2}} = \frac{3}{2} \, t^{1/2}$ provided $t > 0$.
\end{solution}

\end{enumerate}

\end{problem}

 $g'(t) = \lim_{\Delta t \rightarrow 0} \frac{a}{\sqrt{at+a\Delta t+b} + \sqrt{at+b}} = \frac{a}{2 \sqrt{at+b}}$.

We are told $a \neq 0$.  In order for the square roots to be happy, we assume $at + b > 0$.  Hence,   $t >  - \frac{b}{a}$ if $a>0$ and $t < -  \frac{b}{a}$ if $a<0$. 
\end{solution}

\end{enumerate}

\begin{enumerate}

\begin{solution}
$f$ is continuous at $x=0$ since $\lim_{x \rightarrow 0} f(x) = \lim_{x \rightarrow 0} |x| = 0 = |0| = f(0)$.
\end{solution}

 $\lim_{h \rightarrow 0^{-}} \frac{f(h) - f(0)}{h} = \lim_{h \rightarrow 0^{-}} \frac{-h}{h}  =-1$

$\lim_{h \rightarrow 0^{+}} \frac{f(h) - f(0)}{h} = \lim_{h \rightarrow 0^{+}} \frac{h}{h}  = 1$
\end{solution}

 Near $x = 0$, the graph of $y = f(x)$ looks like a `$\vee$' shape\footnote{We say the graph of $f$ has a  \index{corner}\textbf{corner} at $(0,0)$ since we have two different, but finite slopes meeting at a point.} consisting of a line of slope $-1$ to the left of $x=0$ and a line with a slope of $+1$ to the right of $x = 0$.
\end{solution}

\textbf{HINT:}  Consider the two cases $x > 0$ and $x < 0$ \ldots

 \begin{solution}
If $x<0$, $f'(x) =  lim_{h \rightarrow 0} \frac{|x+h| - |x|}{h} = -1$.

If $x>0$, $f'(x) =  \lim_{h \rightarrow 0} \frac{|x+h| - |x|}{h} = 1$.
 \end{solution}


\end{enumerate}

This is the more formal way to describe all of the `unusual steepness' we saw back in Chapter \ref{RootRadicalPowerFunctions}.}  Since $g$ is increasing through $(0,0)$, the `slope' of this vertical line could be seen as $+ \infty$.

\end{solution}
        
\end{enumerate}

$\lim_{\Delta x \rightarrow 0^{+}} \frac{h(\Delta x) - h(0)}{\Delta x} = \lim_{\Delta x \rightarrow 0^{+}} \frac{1}{(\Delta x)^{\frac{1}{3}}} = \infty$

\end{solution}

\begin{problem}

 Recall from Exercise \ref{JasonHammerExercise1} in Section \ref{QuadraticFunctions} that Carl's friend Jason participates in the Highland Games. In one event, the hammer throw, the height $h(t)$ in feet of the hammer above the ground $t$ seconds after Jason lets it go is modeled by the function $h(t) = -16t^2 +  22.08t + 6$. 

\begin{enumerate}

\item  Find and simplify a formula for the velocity of the hammer, $v(t) = h'(t) = \lim_{\Delta t \rightarrow 0} \frac{h(t + \Delta t) - h(t)}{\Delta t}$.

\begin{solution}
$v(t) = \lim_{\Delta t \rightarrow 0} \frac{h(t + \Delta t) - h(t)}{\Delta t} = \lim_{\Delta t \rightarrow 0} \frac{-16(\Delta t)^2 -32 t \, \Delta t + 22.08 \Delta t}{\Delta t}  = -32t + 22.08$.
\end{solution}

\item  Find and interpret $v(0)$.

\begin{solution}
$v(0) = -32(0) + 22.08 = 22.08$. This means initially (when Jason lets go of the hammer), the hammer is traveling upwards at $22.08$ feet per second.
\end{solution}

\item  Solve $v(t) = 0$ and interpret.

\begin{solution}

\begin{enumerate}

\begin{solution}

\end{solution}

\begin{solution}
We see that $C'(75)$ is numerically close to $MC(75)$ but the former is a rate of change (measured in dollars per system) where the latter is a change (measured in dollars).  Note that if we set  $h = 1$ in the difference quotient:  \[\frac{C(75+h) - C(75)}{h} = \frac{C(75+1) - C(75)}{1} = C(76) - C(75)\] we see these two quantities can be used to approximate each other.
\end{solution}

\end{enumerate}
