\begin{enumerate}

\item $\frac{A}{x + 4} + \frac{B}{(x + 4)^{2}} + \frac{C}{(x + 4)^{3}} + \frac{D}{(x + 4)^{4}} + \frac{E}{(x + 4)^{5}} + \frac{Fx + G}{x^{2} + 1} + \frac{Hx + I}{(x^{2} + 1)^{2}}$
\item $\frac{A}{x} + \frac{B}{4x - 1} + \frac{C}{(4x - 1)^{2}} + \frac{Dx + E}{x^{2} + 5} + \frac{Fx + G}{9x^{2} + 16}$


\item $\frac{2x}{x^{2} - 1} = \frac{1}{x + 1} + \frac{1}{x - 1}$
\item $\frac{-7x + 43}{3x^{2} + 19x - 14}= \frac{5}{3x - 2} - \frac{4}{x + 7}$


\item $\frac{11z^{2} - 5z - 10}{5z^{3} - 5z^{2}} = \frac{3}{z} + \frac{2}{z^{2}} - \frac{4}{5(z - 1)}$
\item $\frac{-2z^{2} + 20z - 68}{z^{3} + 4z^{2} + 4z + 16} = -\frac{9}{z + 4} + \frac{7z - 8}{z^{2} + 4}$


\item $\frac{-s^{2} + 15}{4s^{4} + 40s^{2} + 36} = \frac{1}{2(s^{2} + 1)} - \frac{3}{4(s^{2} + 9)}$
\item $\frac{-21s^{2} + s - 16}{3s^{3} + 4s^{2} - 3s + 2} = -\frac{6}{s + 2} - \frac{3s + 5}{3s^{2} - 2s + 1}$


\item $\frac{5x^{4} - 34x^{3} + 70x^{2} - 33x - 19}{(x - 3)^{2}} = 5x^{2} - 4x + 1 + \frac{9}{x - 3} - \frac{1}{(x - 3)^{2}}$
\item $\frac{x^{6} + 5x^{5} + 16x^{4} + 80x^{3} - 2x^{2} + 6x - 43}{x^{3} + 5x^{2} + 16x + 80} = x^{3} + \frac{x + 1}{x^{2} + 16} - \frac{3}{x + 5}$


\item $\frac{-7z^{2} - 76z - 208}{z^{3} + 18z^{2} + 108z + 216} = -\frac{7}{z + 6} + \frac{8}{(z + 6)^{2}} - \frac{4}{(z + 6)^{3}}$
\item $\frac{-10z^{4} + z^{3} - 19z^{2} + z - 10}{z^{5} + 2z^{3} + z} = -\frac{10}{z} + \frac{1}{z^{2} + 1} + \frac{z}{(z^{2} + 1)^{2}}$


\item $\frac{4s^{3} - 9s^{2} + 12s + 12}{s^{4} - 4s^{3} + 8s^{2} - 16s + 16}= \frac{1}{s - 2} + \frac{4}{(s - 2)^{2}} + \frac{3s + 1}{s^{2} + 4}$
\item $\frac{2s^{2} + 3s + 14}{(s^{2} + 2s + 9)(s^{2} + s + 5)} = \frac{1}{s^{2} + 2s + 9} + \frac{1}{s^{2} + s + 5}$

\item $R(z) = \frac{4}{z^4-1} = \frac{1}{z-1} - \frac{1}{z+1} + \frac{i}{z-i} -  \frac{i}{z+i} $

\end{enumerate}
